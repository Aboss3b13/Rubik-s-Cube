private MeshGeometry3D CreateBeveledBox(double size, double bevel)
{
    var mesh = new MeshGeometry3D();
    double s = size / 2.0;
    double d = s - bevel;
    mesh.Positions = new Point3DCollection {
        new Point3D( s, -d, -d ),
        new Point3D( s, d, -d ),
        new Point3D( s, d, d ),
        new Point3D( s, -d, d ),
        new Point3D( -s, -d, d ),
        new Point3D( -s, d, d ),
        new Point3D( -s, d, -d ),
        new Point3D( -s, -d, -d ),
        new Point3D( -d, s, -d ),
        new Point3D( d, s, -d ),
        new Point3D( d, s, d ),
        new Point3D( -d, s, d ),
        new Point3D( -d, -s, d ),
        new Point3D( d, -s, d ),
        new Point3D( d, -s, -d ),
        new Point3D( -d, -s, -d ),
        new Point3D( -d, -d, s ),
        new Point3D( d, -d, s ),
        new Point3D( d, d, s ),
        new Point3D( -d, d, s ),
        new Point3D( -d, d, -s ),
        new Point3D( d, d, -s ),
        new Point3D( d, -d, -s ),
        new Point3D( -d, -d, -s ),
    };
    mesh.TriangleIndices = new Int32Collection {
        0,1,2, 0,2,3,
        4,5,6, 4,6,7,
        11,10,9, 11,9,8,
        15,14,13, 15,13,12,
        16,17,18, 16,18,19,
        20,21,22, 20,22,23,
        0,22,21, 0,21,1,
        1,9,10, 1,10,2,
        2,18,17, 2,17,3,
        3,13,14, 3,14,0,
        4,16,19, 4,19,5,
        5,11,8, 5,8,6,
        6,20,23, 6,23,7,
        7,15,12, 7,12,4,
        11,19,18, 11,18,10,
        8,9,21, 8,21,20,
        12,13,17, 12,17,16,
        15,23,22, 15,22,14,
        2,10,18,
        1,21,9,
        3,17,13,
        0,14,22,
        5,19,11,
        6,8,20,
        4,12,16,
        7,23,15,
    };
    return mesh;
}
